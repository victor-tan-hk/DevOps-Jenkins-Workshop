pipeline {

    // Run the pipeline on any available Jenkins agent 
    // capable of accepting the job.
    //
    // In simplest case, this will be the Jenkins controller 
    // which is also configured to run builds if there is 
    // no separate agent

    agent any

    /*
     * Environment variables are accessible throughout the Pipeline.
     *
     * Adjust these values accordingly 
     */
    environment {

        /*
        * Place the URL for your remote repo (GitHub, BitBucket) here
        */
        REMOTE_REPO_URL = 'https://github.com/victor-tan-hk/PortableCliApp.git'


        SOLUTION_FILE = 'PortableCliApp.slnx'

        APP_PROJECT =
            'src/PortableCliApp/PortableCliApp.csproj'

        TEST_PROJECT =
            'tests/PortableCliApp.Tests/PortableCliApp.Tests.csproj'

        // ${WORKSPACE} typically refers to the absolute filesystem path of 
        // the workspace allocated to the current Jenkins build.
        // 
        // Typically this will be
        // /var/lib/jenkins/workspace/Jenkins-job-name

        PUBLISH_DIRECTORY =
            "${WORKSPACE}/artifacts/publish"

        TEST_RESULTS_DIRECTORY =
            "${WORKSPACE}/artifacts/test-results"
            
    }


    stages {

        stage('SCM Checkout') {
            steps {

                git "$REMOTE_REPO_URL"
            
            }
        }

        stage('Clean') {
            steps {
                /*
                 * Remove build artifacts left in the current workspace
                 * from a previous run.
                 *
                 * Remove published application files
                 * generated by previous pipeline runs.
                 *
                 */
                sh '''

                    rm -rf "${PUBLISH_DIRECTORY}"

                '''
            }
        }

        stage('Restore') {
            steps {
                /*
                 * Download the NuGet packages required by the application
                 * and test projects.
                 */
                sh '''
                    dotnet restore "$SOLUTION_FILE"
                '''
            }
        }

        stage('Build') {
            steps {
                /*
                 * Build the application and test project in Release mode.
                 *
                 * --no-restore prevents a second implicit package restore
                 * because the Restore stage has already completed it.
                 */
                sh '''
                    dotnet build "$SOLUTION_FILE" \
                        --configuration Release \
                        --no-restore
                '''
            }
        }

        stage('Test') {

            steps {
                /*
                 * Execute all unit tests.
                 * --no-build uses the assemblies produced by the Build stage.
                 */
                sh '''

                    dotnet test "$SOLUTION_FILE" \
                        --configuration Release \
                        --no-build \
                        --no-restore
       
                '''
            }
        }

        stage('Publish') {
            steps {
                /*
                 * Publish a framework-dependent Linux application.
                 *
                 * --no-build reuses the successful Release build.
                 */
                sh '''

                    dotnet publish "$APP_PROJECT" \
                        --configuration Release \
                        --no-build \
                        --output "${PUBLISH_DIRECTORY}"

                    echo "Published files:"

                    find "${PUBLISH_DIRECTORY}" \
                        -maxdepth 2 \
                        -type f \
                        -print
                '''
            }
        }

        stage('Deploy') {
            steps {
                /*
                 * Run the published CLI application once.
                 *
                 * This is a simple deployment verification or smoke test
                 * confirming that the published application can start.
                 */
                sh '''

                    dotnet "${PUBLISH_DIRECTORY}/PortableCliApp.dll"

                '''
            }
        }
    }

}
